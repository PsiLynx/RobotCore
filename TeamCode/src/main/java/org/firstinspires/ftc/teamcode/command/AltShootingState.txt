package org.firstinspires.ftc.teamcode.command

import org.firstinspires.ftc.teamcode.command.internal.Command
import org.firstinspires.ftc.teamcode.controller.VaState
import org.firstinspires.ftc.teamcode.geometry.Pose2D
import org.firstinspires.ftc.teamcode.subsystem.Flywheel
import org.firstinspires.ftc.teamcode.subsystem.Hood
import org.firstinspires.ftc.teamcode.subsystem.internal.Subsystem
import java.util.function.Supplier
import kotlin.math.PI
import kotlin.math.pow

class AltShootingState(val pos: Supplier<Pose2D>): Command() {
    override val requirements = mutableSetOf<Subsystem<*,*>>(Hood, Flywheel)
    override fun execute() {
        Shooter.io.flywheelUsingFeedback = true
        val dist = (
            pos.get().vector - goalPose.groundPlane
        ).mag

        Shooter.io.hoodTargetAngle = PI/2 - (
            if(dist > 20) {(
                -(3.30185*10.0.pow(-8)) * dist.pow(4)
                +0.00000849965 * dist.pow(3)
                -0.000756316 * dist.pow(2)
                +0.021754 * dist
                +1.1

            )} else  1.36
        )
        Shooter.io.flywheelTargetState = VaState(
            Shooter.io.flywheelGetVel(
                PI/2 - Shooter.io.hoodTargetAngle,
                dist
            ),
            0.0
        )

    }

    override fun end(interrupted: Boolean) {
        Shooter.io.flywheelMotors.forEach { it.power = 0.0 }
        Shooter.io.flywheelUsingFeedback = false
        Shooter.io.hoodTargetAngle = Shooter.io.hoodMinAngle
        Shooter.io.hoodUpdate()
    }
}
